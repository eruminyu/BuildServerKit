📋 신규 PC Perforce 서버 초기 명령어 빠른 참조

                
              

              
# 1) 한글 Unicode 활성화 (최초 1회)
net stop Perforce && p4d -r "C:\Perforce\Server" -xi && net start Perforce
p4 set P4PORT=localhost:1666 && p4 set P4CHARSET=utf8

# 2) Read-Only 봇(p4bot) + 무기한 티켓
p4 -u admin user -f p4bot
p4 -u admin group automation_bots   # Timeout: unlimited, Users: p4bot
p4 -u admin protect                 # 하단 추가: read group automation_bots * //...
p4 -p localhost:1666 -u p4bot login
